const { chromium }=require(process.cwd()+'/node_modules/playwright');const fs=require('fs');
const out=process.argv[2];if(!out)throw new Error('Pass an output directory argument');fs.mkdirSync(out,{recursive:true});
(async()=>{const browser=await chromium.launch({channel:'chrome',headless:true});const page=await browser.newPage({viewport:{width:1440,height:1000}});const errors=[];page.on('pageerror',e=>errors.push(e.message));await page.goto('http://127.0.0.1:3000/landing-editorial',{waitUntil:'networkidle'});await page.evaluate(()=>document.fonts.ready);await page.getByRole('button',{name:'Necessary only',exact:true}).click();
 const states=[];async function state(label){const d=await page.evaluate(()=>{const s=document.querySelector('#how-it-works'),p=s.querySelector('[data-walkthrough-pin]');return{y:scrollY,hash:location.hash,nav:!!s.closest('[data-section-navigation]'),compact:s.hasAttribute('data-navigation-compact'),anchoring:document.documentElement.style.overflowAnchor,pinPosition:getComputedStyle(p).position,pinPaddingBottom:getComputedStyle(p).paddingBottom,active:s.querySelector('[aria-selected="true"]')?.innerText,qualityTop:document.querySelector('#quality').getBoundingClientRect().top,pricingTop:document.querySelector('#pricing').getBoundingClientRect().top};});states.push({label,...d});}
 async function top(){await page.evaluate(()=>scrollTo({top:0,behavior:'instant'}));await page.waitForTimeout(80);}
 await top();await page.locator('header a[href="#quality"]').first().click();await page.waitForTimeout(110);await page.locator('header a[href="#pricing"]').first().click();await page.waitForTimeout(1100);await state('repeat-quality-then-pricing');
 await top();await page.locator('header a[href="#quality"]').first().click();await page.waitForTimeout(80);await page.keyboard.press('PageDown');await page.waitForTimeout(400);await state('cancel-page-down');
 await top();await page.locator('header a[href="#quality"]').first().click();await page.waitForTimeout(80);await page.setViewportSize({width:1024,height:900});await page.waitForTimeout(350);await state('cancel-desktop-resize');
 await page.setViewportSize({width:1440,height:1200});await top();await page.locator('header a[href="#quality"]').first().click();await page.waitForTimeout(80);await page.setViewportSize({width:390,height:844});await page.waitForTimeout(350);await state('cancel-to-mobile');
 await page.getByRole('button',{name:'Open navigation',exact:true}).click();await page.locator('#editorial-mobile-nav a[href="#quality"]').click();await page.waitForTimeout(1100);await state('mobile-quality');
 await page.setViewportSize({width:1440,height:1200});await page.waitForTimeout(120);for(const progress of[0,.5,1,.5,0]){await page.evaluate(progress=>{const s=document.querySelector('#how-it-works'),p=s.querySelector('[data-walkthrough-pin]');const a=scrollY+s.getBoundingClientRect().top-parseFloat(getComputedStyle(p).top);scrollTo({top:a+(s.offsetHeight-p.offsetHeight)*progress,behavior:'instant'});},progress);await page.waitForTimeout(70);await state('manual-'+progress);}
 // Keyboard focus is transferred only on completed keyboard navigation.
 await page.locator('header a[href="#quality"]').first().focus();await page.keyboard.press('Enter');await page.waitForTimeout(1100);await state('keyboard-completed');const focused=await page.evaluate(()=>document.activeElement.id);
 const result={states,focused,errors};fs.writeFileSync(out+'/lifecycle-manual.json',JSON.stringify(result,null,2));console.log(JSON.stringify(result,null,2));await browser.close();})().catch(e=>{console.error(e);process.exitCode=1});
